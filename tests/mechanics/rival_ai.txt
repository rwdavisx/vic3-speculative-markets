namespace = spec_mechanics_rival_ai
spec_mechanics_rival_ai.1 = { type = country_event hidden = yes immediate = {
# The published 1.1 AI reviews contest position and age every quarter.
# Keep the two competitors equal at first, and make lure the only offensive move available.
every_country = {
    clear_variable_list = spec_rv_chosen clear_variable_map = spec_rv_pts
    set_variable = { name = spec_rv_ai_wait days = 365 }
}
every_country = {
    limit = { OR = { this = c:FRA this = c:PRU } }
    set_variable = { name = spec_exch_stand value = 1 }
    set_variable = { name = spec_centre_score value = 100 }
    set_variable = { name = spec_centre_trust value = 60 }
    set_variable = { name = spec_debt value = 0 }
    remove_variable = spec_bc_n
    remove_variable = spec_rv_focus
    remove_variable = spec_rv_hit_recent
    remove_variable = spec_panic
    clear_variable_list = spec_rv_cd_lure
    clear_variable_list = spec_rv_cd_shore
    clear_variable_list = spec_rv_shield
    add_treasury = 100000000
}
c:PRU = { save_scope_as = target }
c:FRA = {
    add_to_variable_list = { name = spec_rv_chosen target = c:PRU }
    add_to_variable_list = { name = spec_rv_cd_raid target = c:PRU }
    add_to_variable_list = { name = spec_rv_cd_press target = c:PRU }
    spec_rv_ai_act = yes
    if = { limit = { NOT = { has_local_variable = spec_rv_ai_moved } } debug_log = "HOTFIX PASS early_tied_contest_waits" }
    else = { debug_log = "HOTFIX FAIL early_tied_contest_waits" }
    add_to_variable_list = { name = spec_rv_age_3 target = c:PRU }
    spec_rv_ai_act = yes
    if = { limit = { is_target_in_variable_list = { name = spec_rv_cd_lure target = c:PRU } } debug_log = "HOTFIX PASS late_tied_contest_acts" }
    else = { debug_log = "HOTFIX FAIL late_tied_contest_acts" }
    set_variable = spec_rv_hit_recent
    set_variable = { name = spec_rv_hit_by value = c:PRU }
    spec_rv_ai_act = yes
    if = { limit = { has_local_variable = spec_rv_ai_moved NOT = { is_target_in_variable_list = { name = spec_rv_shield target = c:PRU } } } debug_log = "HOTFIX PASS ai_once_per_rivalry_per_call" }
    else = { debug_log = "HOTFIX FAIL ai_once_per_rivalry_per_call" }
    remove_local_variable = spec_rv_ai_moved
    spec_rv_ai_act = yes
    if = { limit = { is_target_in_variable_list = { name = spec_rv_shield target = c:PRU } is_target_in_variable_list = { name = spec_rv_cd_shore target = c:PRU } } debug_log = "HOTFIX PASS ai_shores_up_after_hit" }
    else = { debug_log = "HOTFIX FAIL ai_shores_up_after_hit" }
    remove_local_variable = spec_rv_ai_moved
    remove_variable = spec_rv_hit_recent
    remove_variable = spec_rv_hit_by
    clear_variable_list = spec_rv_cd_lure
    set_variable = { name = spec_centre_score value = 10000 }
    spec_rv_ai_act = yes
    if = { limit = { NOT = { has_local_variable = spec_rv_ai_moved } } debug_log = "HOTFIX PASS decisive_leader_holds" }
    else = { debug_log = "HOTFIX FAIL decisive_leader_holds" }
    c:PRU = { remove_variable = spec_rv_lure }  # the earlier lure has ended
    set_variable = { name = spec_centre_score value = 1 }
    clear_variable_list = spec_rv_age_3
    c:PRU = { save_scope_as = target }
    set_variable = { name = spec_debt value = 0 }  # isolate this look from credit raised by the previous lure
    spec_rv_ai_act = yes
    if = { limit = { is_target_in_variable_list = { name = spec_rv_cd_lure target = c:PRU } } debug_log = "HOTFIX PASS losing_contest_acts" }
    else = { debug_log = "HOTFIX FAIL losing_contest_acts" }
    clear_variable_list = spec_rv_cd_lure
    set_variable = spec_rv_ai_seen
    remove_variable = spec_rv_ai_wait
    add_to_variable_list = { name = spec_rv_chosen target = c:AUS }
}
c:AUS = { remove_variable = spec_centre_score }
c:RUS = { remove_variable = spec_centre_score add_to_variable_list = { name = spec_rv_chosen target = c:FRA } }
spec_rv_ai = yes
    if = { limit = { c:FRA = { has_variable = spec_rv_ai_wait NOT = { is_target_in_variable_list = { name = spec_rv_cd_lure target = c:AUS } } NOT = { is_target_in_variable_list = { name = spec_rv_cd_lure target = c:RUS } } } } debug_log = "HOTFIX PASS ai_skips_unscored_competitors" }
    else = { debug_log = "HOTFIX FAIL ai_skips_unscored_competitors" }
    if = { limit = { c:GBR = { NOT = { var:spec_rv_hit_by ?= c:FRA } NOT = { var:spec_rv_hit_by ?= c:PRU } } } debug_log = "HOTFIX PASS ai_never_acts_unpaired" }
    else = { debug_log = "HOTFIX FAIL ai_never_acts_unpaired" }
c:FRA = { add_to_variable_list = { name = spec_rv_cd_raid target = c:PRU days = 365 } c:PRU = { save_scope_as = target } spec_rv_raid = yes }
if = { limit = { c:PRU = { var:spec_rv_hit = 1 var:spec_rv_hit_by = c:FRA } } debug_log = "HOTFIX PASS retaliation_ignores_cooldown" }
else = { debug_log = "HOTFIX FAIL retaliation_ignores_cooldown" }
debug_log = "HOTFIX SUITE COMPLETE"
} }

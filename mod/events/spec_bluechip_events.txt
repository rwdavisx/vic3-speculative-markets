namespace = spec_bluechip

spec_bluechip.1 = {
	type = country_event
	placement = root
	title = spec_bluechip.1.t
	desc = spec_bluechip.1.d
	flavor = spec_bluechip.1.f
	duration = 3
	event_image = {
		trigger = { spec_pic_mideast = yes }
		video = "unspecific_signed_contract"
	}
	event_image = {
		trigger = { spec_pic_africa = yes }
		video = "unspecific_signed_contract"
	}
	event_image = {
		trigger = { spec_pic_asia = yes }
		video = "asia_westerners_arriving"
	}
	event_image = {
		trigger = { spec_pic_latam = yes }
		video = "unspecific_signed_contract"
	}
	event_image = {
		trigger = { spec_pic_west = yes }
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	trigger = {
		any_company = { has_variable = spec_bc_new }
	}

	immediate = {
		random_company = {
			limit = { has_variable = spec_bc_new }
			save_scope_as = spec_bc_co
		}
		scope:spec_bc_co = { remove_variable = spec_bc_new }
	}

	option = {
		name = spec_bluechip.1.a
		default_option = yes
		custom_tooltip = spec_bluechip.1.a.tt
		hidden_effect = {
			if = {
				limit = { has_variable = spec_centre_trust }
				set_variable = { name = spec_centre_trust value = { value = var:spec_centre_trust add = 2 max = 120 } }
			}
		}
		ai_chance = {
			base = 60
		}
	}

	option = {
		name = spec_bluechip.1.b
		custom_tooltip = spec_bluechip.1.b.tt
		hidden_effect = {
			add_investment_pool = spec_bc_shares
			scope:spec_bc_co = { set_variable = { name = spec_bc_fragile days = 1825 } }
		}
		ai_chance = {
			base = 40
		}
	}
}

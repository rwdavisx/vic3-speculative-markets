# the investment modifier of the confidence band the country is in, and no other
spec_update_band = {
	if = {
		limit = { NOT = { has_modifier = spec_conf_panic } var:spec_conf < 20 NOT = { has_modifier = spec_panic } }
		add_modifier = { name = spec_conf_panic }
	}
	else_if = {
		limit = { has_modifier = spec_conf_panic NOT = { AND = { var:spec_conf < 20 NOT = { has_modifier = spec_panic } } } }
		remove_modifier = spec_conf_panic
	}
	if = {
		limit = { NOT = { has_modifier = spec_conf_panic_held } var:spec_conf < 20 has_modifier = spec_panic }
		add_modifier = { name = spec_conf_panic_held }
	}
	else_if = {
		limit = { has_modifier = spec_conf_panic_held NOT = { AND = { var:spec_conf < 20 has_modifier = spec_panic } } }
		remove_modifier = spec_conf_panic_held
	}
	if = {
		limit = { NOT = { has_modifier = spec_conf_wary } var:spec_conf >= 20 var:spec_conf < 40  }
		add_modifier = { name = spec_conf_wary }
	}
	else_if = {
		limit = { has_modifier = spec_conf_wary NOT = { AND = { var:spec_conf >= 20 var:spec_conf < 40  } } }
		remove_modifier = spec_conf_wary
	}
	if = {
		limit = { NOT = { has_modifier = spec_conf_steady } var:spec_conf >= 40 var:spec_conf < 60  }
		add_modifier = { name = spec_conf_steady }
	}
	else_if = {
		limit = { has_modifier = spec_conf_steady NOT = { AND = { var:spec_conf >= 40 var:spec_conf < 60  } } }
		remove_modifier = spec_conf_steady
	}
	if = {
		limit = { NOT = { has_modifier = spec_conf_buoyant } var:spec_conf >= 60 var:spec_conf < 80  }
		add_modifier = { name = spec_conf_buoyant }
	}
	else_if = {
		limit = { has_modifier = spec_conf_buoyant NOT = { AND = { var:spec_conf >= 60 var:spec_conf < 80  } } }
		remove_modifier = spec_conf_buoyant
	}
	if = {
		limit = { NOT = { has_modifier = spec_conf_mania } var:spec_conf >= 80  }
		add_modifier = { name = spec_conf_mania }
	}
	else_if = {
		limit = { has_modifier = spec_conf_mania NOT = { AND = { var:spec_conf >= 80  } } }
		remove_modifier = spec_conf_mania
	}
}

spec_start_panic = {
	# months left, counted down monthly; the one clock for the panic
	set_variable = { name = spec_panic value = spec_panic_months_trust }
	set_variable = { name = spec_panic_len value = var:spec_panic }
	set_variable = { name = spec_panic_elapsed value = 0 }
	# a panic that starts high up still falls to the bottom, and the tab shows the fall. Just under the line: on it is the Wary band
	if = {
		limit = { var:spec_conf >= 25 }
		set_variable = { name = spec_c_drop value = { value = 24.9 subtract = var:spec_conf } }
		change_variable = { name = spec_c_total add = var:spec_c_drop }
		set_variable = { name = spec_conf value = 24.9 }
		if = {
			limit = { has_variable = spec_hist1 }
			set_variable = { name = spec_hist1 value = var:spec_conf }
		}
	}
	if = {
		limit = { spec_crunch_if > 0 }
		if = {
			limit = { has_variable = spec_crunch }
			remove_modifier = spec_credit_crunch
		}
		# the bailout's button comes back only for Bad Debts larger than the ones it bought up
		if = {
			limit = { spec_crunch_if > spec_crunch_sv }
			remove_variable = spec_bailed
		}
		set_variable = { name = spec_crunch value = spec_crunch_if }
		if = {
			limit = { NOT = { has_variable = spec_crunch_bite } }
			set_variable = { name = spec_crunch_bite value = 0 }
		}
		change_variable = { name = spec_crunch_bite add = 3 }
if = {
	limit = { var:spec_crunch_bite > var:spec_crunch }
	set_variable = { name = spec_crunch_bite value = var:spec_crunch }
}
add_modifier = {
	name = spec_credit_crunch
	multiplier = var:spec_crunch_bite
}

	}
	set_variable = { name = spec_ld_recent value = yes days = 365 }
	if = {
		limit = {
			has_variable = spec_ld
			has_global_variable = spec_ld_phase
		}
		spec_ld_apply = yes
	}
	add_modifier = { name = spec_panic }
	add_modifier = {
		name = spec_panic_crunch
		multiplier = spec_alloc_half
	}
	add_modifier = {
		name = spec_panic_relief
		multiplier = spec_panic_relief_week
	}
	spec_update_band = yes
	# ruined investors turn on the government; workers radicalize through vanilla's loss of living standards
	add_radicals = { value = 0.1 pop_type = capitalists }
	add_radicals = { value = 0.05 pop_type = aristocrats }
	add_radicals = { value = 0.05 pop_type = shopkeepers }
	# banks call in part of the credit they lent
	set_variable = { name = spec_recall_balance value = 0 }
	if = { limit = { has_variable = spec_debt } set_variable = { name = spec_recall_balance value = var:spec_debt } }
	set_variable = { name = spec_recall_requested value = { value = var:spec_recall_balance multiply = 0.5 } }
	set_variable = { name = spec_recall_pool_share value = spec_claw_share }
	set_variable = { name = spec_clawed value = spec_clawback }
	add_investment_pool = { value = var:spec_clawed multiply = -1 }
	change_variable = { name = spec_debt subtract = var:spec_clawed }
	set_variable = { name = spec_centre_calm value = 0 }  # centres.py: a rescued panic can end before its monthly pass sees it
	if = {
		limit = { has_variable = spec_centre_trust }
		set_variable = { name = spec_centre_trust value = { value = var:spec_centre_trust subtract = { value = var:spec_centre_trust multiply = spec_exch_panic_cut multiply = spec_exch_loss } min = 0 } }
	}
	set_variable = { name = spec_debt value = { value = var:spec_debt add = var:spec_clawed multiply = 0.5 } }
	if = {
		limit = { has_variable = spec_lev_old }
		change_variable = { name = spec_lev_old multiply = 0.5 }
	}
	if = {
		limit = { has_variable = spec_pledge }
		if = {
			limit = {
				spec_can_rescue = yes
				in_default = no
				gold_reserves >= spec_rescue_cost
			}
			spec_rescue = yes
		}
		else = {
			set_variable = { name = spec_pledge_unmet days = 30 }
			if = { limit = { in_default = yes } set_variable = { name = spec_pledge_reason value = 2 days = 30 } }
			else_if = { limit = { NOT = { has_technology_researched = banking } } set_variable = { name = spec_pledge_reason value = 3 days = 30 } }
			else_if = { limit = { gold_reserves < spec_rescue_cost }
				set_variable = { name = spec_pledge_reason value = 1 days = 30 }
				set_variable = { name = spec_pledge_required value = spec_rescue_cost days = 30 }
				set_variable = { name = spec_pledge_shortfall value = { value = spec_rescue_cost subtract = gold_reserves } days = 30 }
			}
		}
	}
	trigger_event = { id = spec_markets.2 popup = yes }
	# contagion: countries tied to this one lose confidence, the closer the tie and the bigger this economy
	# compared with theirs, the more
	set_variable = { name = spec_src_gdp value = gdp }
	if = {
		limit = { NOT = { has_variable = spec_caught } }
		spec_every_tied = { EFFECT = spec_catch_panic }
		spec_notify_tied = { TYPE = spec_tied_panic }
	}
	spec_capital_flight = yes
	spec_panic_movements = yes
}

# runs $EFFECT$ = { WEIGHT = tie weight } on each country tied to root, once per tie, so a country with two ties
# gets it twice. scope:spec_tie_src is the country whose ownership spec_owned_share measures
spec_every_tied = {
	root = { save_scope_as = spec_tie_src }
	market ?= {
		every_scope_country = {
			limit = { NOT = { this = root } }
			$EFFECT$ = { WEIGHT = spec_tie_market }
		}
	}
	every_subject_or_below = {
		$EFFECT$ = { WEIGHT = 0.8 }
	}
	every_overlord_or_above = {
		$EFFECT$ = { WEIGHT = 0.8 }
	}
	every_country = {
		limit = {
			NOT = { this = root }
			any_scope_treaty = {
				binds = root
				any_scope_article = { has_type = foreign_investment_rights }
			}
		}
		$EFFECT$ = { WEIGHT = spec_tie_investment }
	}
	every_country = {
		limit = {
			NOT = { this = root }
			spec_owned_share >= 0.1
		}
		$EFFECT$ = { WEIGHT = spec_tie_owned }
	}
	every_market = {
		limit = {
			NOT = { this = root.market }
			market_trade_reliance = { target = root.market value > 0.2 }
		}
		every_scope_country = {
			limit = { spec_no_embargo_with_root = yes }
			$EFFECT$ = { WEIGHT = spec_tie_trade }
		}
	}
}

# on a country tied to root, which has just panicked: keep the biggest hit until its next monthly update
spec_catch_panic = {
	if = {
		limit = {
			has_variable = spec_conf
			gdp > 0
		}
		set_variable = {
			name = spec_foreign_hit
			value = {
				value = root.var:spec_src_gdp
				divide = gdp
				max = 1
				multiply = 20
				multiply = $WEIGHT$
				multiply = root.spec_reach
				multiply = spec_foreign_loss_mult
				multiply = spec_world_shelter
				multiply = spec_bc_shelter
			}
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable = spec_foreign_pending }
					var:spec_foreign_hit > var:spec_foreign_pending
				}
			}
			set_variable = { name = spec_foreign_pending value = var:spec_foreign_hit }
			set_variable = { name = spec_foreign_src value = root }
			# the largest hit's parts, for the calculator
			set_variable = { name = spec_fh_size value = { value = root.var:spec_src_gdp divide = gdp max = 1 } }
			set_variable = { name = spec_fh_tie value = $WEIGHT$ }
			set_variable = { name = spec_fh_reach value = root.spec_reach }
		}
		if = {
			limit = {
				NOT = { has_variable = spec_fp1 }
			}
			set_variable = { name = spec_fp1 value = root }
		}
		else_if = {
			limit = {
				NOT = { var:spec_fp1 ?= root }
				NOT = { has_variable = spec_fp2 }
			}
			set_variable = { name = spec_fp2 value = root }
		}
		else_if = {
			limit = {
				NOT = { var:spec_fp1 ?= root }
				NOT = { var:spec_fp2 ?= root }
				NOT = { has_variable = spec_fp3 }
			}
			set_variable = { name = spec_fp3 value = root }
		}

		remove_variable = spec_foreign_hit
	}
}

# for the tab, on a player: the tied country whose panic would cost the most, among those near one (Wary and
# below) or in a mania. The ties mirror spec_start_panic's, seen from the country that would catch it.
spec_find_risk_src = {
	remove_variable = spec_risk_src
	remove_variable = spec_risk_src_tie
	set_variable = { name = spec_risk_src_hit value = 0 }
	market ?= {
		every_scope_country = {
			limit = { NOT = { this = root } }
			spec_rate_risk_src = { WEIGHT = root.spec_tie_market TIE = 1 }
		}
	}
	every_subject_or_below = {
		spec_rate_risk_src = { WEIGHT = 0.8 TIE = 2 }
	}
	every_overlord_or_above = {
		spec_rate_risk_src = { WEIGHT = 0.8 TIE = 2 }
	}
	every_country = {
		limit = {
			NOT = { this = root }
			any_scope_treaty = {
				binds = root
				any_scope_article = { has_type = foreign_investment_rights }
			}
		}
		spec_rate_risk_src = { WEIGHT = root.spec_tie_investment TIE = 3 }
	}
	every_country = {
		limit = { NOT = { this = root } }
		save_scope_as = spec_tie_src
		if = {
			limit = { root = { spec_owned_share >= 0.1 } }
			spec_rate_risk_src = { WEIGHT = root.spec_tie_owned TIE = 5 }
		}
	}
	every_market = {
		limit = {
			NOT = { this = root.market }
			root.market = { market_trade_reliance = { target = prev value > 0.2 } }
		}
		every_scope_country = {
			limit = { spec_no_embargo_with_root = yes }
			spec_rate_risk_src = { WEIGHT = root.spec_tie_trade TIE = 4 }
		}
	}
	if = {
		limit = { NOT = { has_variable = spec_risk_src } }
		remove_variable = spec_risk_src_hit
	}
}

# on a country tied to root: keep it as root's outside risk if its panic would hit root hardest so far
spec_rate_risk_src = {
	if = {
		limit = {
			has_variable = spec_conf
			NOT = { has_variable = spec_panic }
			NOT = { has_variable = spec_panic_cooldown }
			OR = {
				var:spec_conf < 40
				var:spec_conf >= 80
			}
			root.gdp > 0
		}
		set_variable = {
			name = spec_risk_try
			value = {
				value = gdp
				divide = root.gdp
				max = 1
				multiply = 20
				multiply = $WEIGHT$
				multiply = spec_reach
			}
		}
		if = {
			limit = { var:spec_risk_try > root.var:spec_risk_src_hit }
			# its parts, for the calculator, read here: the tie's weight belongs to this country
			set_variable = { name = spec_risk_w value = $WEIGHT$ }
			set_variable = { name = spec_risk_r value = spec_reach }
			set_variable = { name = spec_risk_s value = { value = gdp divide = root.gdp max = 1 } }
			root = {
				set_variable = { name = spec_risk_src_hit value = prev.var:spec_risk_try }
				set_variable = { name = spec_risk_src value = prev }
				set_variable = { name = spec_risk_src_tie value = $TIE$ }
				set_variable = { name = spec_risk_src_w value = prev.var:spec_risk_w }
				set_variable = { name = spec_risk_src_reach value = prev.var:spec_risk_r }
				set_variable = { name = spec_risk_src_size value = prev.var:spec_risk_s }
			}
			remove_variable = spec_risk_w
			remove_variable = spec_risk_r
			remove_variable = spec_risk_s
		}
		remove_variable = spec_risk_try
	}
}

spec_end_panic = {
	if = {
		limit = { is_player = yes }
		post_notification = spec_panic_over
	}
	if = {
		limit = {
			has_variable = spec_lax
			NOT = { has_variable = spec_rescued }
		}
		set_variable = { name = spec_lax value = { value = var:spec_lax subtract = 0.4 min = 0 } }
	}
	if = {
		limit = {
			has_variable = spec_centre_trust
			NOT = { has_variable = spec_rescued }
			var:spec_panic <= 0
		}
		set_variable = { name = spec_centre_trust value = { value = var:spec_centre_trust subtract = { value = var:spec_centre_trust multiply = 0.5 multiply = spec_exch_loss } min = 0 } }
		set_variable = { name = spec_exch_sound value = 0 }
	}
	remove_variable = spec_panic
	remove_variable = spec_panic_len
	remove_variable = spec_panic_elapsed
	remove_variable = spec_rescued
	remove_modifier = spec_panic
	remove_modifier = spec_panic_crunch
	remove_modifier = spec_panic_relief
	set_variable = { name = spec_panic_cooldown value = 12 }
	spec_update_band = yes
}

# the bank rescue, once per panic, from the tab, the journal entry or the panic event
spec_rescue = {
		custom_tooltip = {
			text = spec_rescue_cost_tt
			add_treasury = { value = spec_rescue_cost multiply = -1 }
		}
		custom_tooltip = {
			text = spec_bank_bailout_tt
			ig:ig_trade_unions ?= {
				add_modifier = {
					name = spec_bank_bailout
					days = short_modifier_time
				}
			}
			ig:ig_rural_folk ?= {
				add_modifier = {
					name = spec_bank_bailout
					days = short_modifier_time
				}
			}
		}

	# the tab's button and the events say what this does in words (SPEC_RESCUE_TT): the swap is not listed
	hidden_effect = {
		# the Industrialists approved of leaving the market alone (the panic event's first option): a rescue ends that
		ig:ig_industrialists ?= { remove_modifier = spec_market_left_alone }
		remove_modifier = spec_panic
		remove_modifier = spec_panic_crunch
		add_modifier = {
			name = spec_panic
			multiplier = 0.5
		}
		add_modifier = {
			name = spec_panic_crunch
			multiplier = spec_alloc_quarter
		}
	}
	set_variable = { name = spec_panic value = { value = var:spec_panic max = spec_rescue_months } }
	# the jump, after the 100 limit, shown in the tab until the next monthly update
	set_variable = { name = spec_c_rescue value = { value = var:spec_conf multiply = -1 } }
	set_variable = { name = spec_conf value = { value = var:spec_conf add = 20 max = 100 } }
	change_variable = { name = spec_c_rescue add = var:spec_conf }
	change_variable = { name = spec_c_total add = var:spec_c_rescue }
	set_variable = spec_rescued
	if = {
		limit = { has_variable = spec_lax }
		set_variable = { name = spec_lax value = { value = var:spec_lax add = 0.25 max = 1 } }
	}
	spec_pick_driver = yes
	spec_update_band = yes
}

					name = spec_bank_bailout

			}
		}
	hidden_effect = {
		# the button's tooltip previews this effect even with no Bad Debts: read nothing that is unset
		if = {
			limit = { has_variable = spec_crunch has_variable = spec_crunch_bite }
			remove_modifier = spec_credit_crunch
			set_variable = { name = spec_crunch value = { value = var:spec_crunch multiply = 0.5 } }

			limit = { has_variable = spec_lax }

		}
	}
}

# the mania event's options: confidence down by AMOUNT, never below 0, shown in the tab as this month's change
spec_cool_conf = {
	# Pending mania and pledge responses must not restart markets after a law change.
	if = {
		limit = { has_variable = spec_conf spec_law_mult > 0 }
		if = {
			limit = { NOT = { has_variable = spec_c_total } }
			set_variable = { name = spec_c_total value = 0 }
		}
		set_variable = { name = spec_c_caution value = { value = var:spec_conf max = $AMOUNT$ multiply = -1 } }
		change_variable = { name = spec_c_total add = var:spec_c_caution }
		set_variable = { name = spec_conf value = { value = var:spec_conf subtract = $AMOUNT$ min = 0 } }
		spec_pick_driver = yes
		spec_update_band = yes
	}

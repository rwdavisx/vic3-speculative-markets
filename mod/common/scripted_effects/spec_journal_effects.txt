# a Treasury letter lets the bank lend beyond the Act's limit: the panic ends soon, at no cost to the treasury, but
# investors learn that the government will step in, and the next bubble grows faster
spec_do_suspend = {
	add_modifier = {
		name = spec_act_suspended
		years = 2
	}
	custom_tooltip = {
		text = spec_suspend_tt
		set_variable = { name = spec_panic value = { value = var:spec_panic max = 2 } }
		set_local_variable = { name = spec_jump_was value = var:spec_conf }
		set_variable = { name = spec_conf value = { value = var:spec_conf add = 15 max = 100 } }
		if = {
			limit = { NOT = { has_variable = spec_c_rescue } }
			set_variable = { name = spec_c_rescue value = 0 }
		}
		set_local_variable = { name = spec_jump value = { value = var:spec_conf subtract = local_var:spec_jump_was } }
		change_variable = { name = spec_c_rescue add = local_var:spec_jump }
		change_variable = { name = spec_c_total add = local_var:spec_jump }
		set_variable = spec_rescued
		if = {
			limit = { has_variable = spec_lax }
			set_variable = { name = spec_lax value = { value = var:spec_lax add = 0.25 max = 1 } }
		}
		set_variable = { name = spec_hazard years = 10 }
		spec_pick_driver = yes
		spec_update_band = yes
	}
}

# after an institution is made: the count for the bar, and Sound Finance once all three stand
spec_reform_done = {
	set_variable = { name = spec_reforms value = spec_reform_count }
	je:je_spec_markets ?= {
		set_bar_progress = { name = spec_finance_bar value = root.var:spec_reforms }
	}
	if = {
		limit = {
			spec_reform_count >= 3
			NOT = { has_variable = spec_sound }
		}
		set_variable = spec_sound
		trigger_event = { id = spec_reform.2 popup = yes }
	}
}

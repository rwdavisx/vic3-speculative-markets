pmr_ai_value_art_academy = {
	value = 1000
	if = { # AI Metternich really wants arts academies
		limit = {
			owner ?= {
				c:AUS ?= this
				has_journal_entry = je_metternich
			}
		}
            add = 500
	}
	if = {
		limit = {
			OR = {
				AND = {
					owner ?= { has_variable = ai_ukraine_renaissance_accelerate }
					state_region = { is_homeland = cu:ukrainian }
				}
				AND = {
					owner ?= { has_variable = ai_belarus_renaissance_accelerate }
					state_region = { is_homeland = cu:byelorussian }
				}
			}
		}
		add = 1000
	}
	if = { # Crown of Perfection agenda - patronage of the arts in the capital
		limit = {
			owner ?= {
				has_journal_entry = je_tp_tsar_crown_of_perfection
				NOT = { has_variable = je_tp_tsar_crown_of_perfection_button_1_complete }
			}
			is_capital = yes
		}
		add = 1000
	}
}

	value = 1000

	# Need 1 level of Power Plants everywhere as its a local good and a prerequisite to enabling electricity PMs to begin with

		limit = {
			NOR = {
				has_building = building_power_plant
				any_scope_building = {
					is_building_type = building_power_plant
					is_under_construction = yes
				}
			}
		}
		add = 4000
	}

	if = {
		limit = {
			owner ?= {
				has_journal_entry = je_five_year_plan
				je:je_five_year_plan ?= {
					has_variable = five_year_plan_electrification_var
				}
				five_year_plan_electrification_goal_met = no
			}
		}
		add = 2500
	}
}

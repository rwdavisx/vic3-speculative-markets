principle_spec_financial_integration_1 = {
	icon = "gfx/interface/icons/spec_markets/principle_financial_integration.dds"
	background = "gfx/interface/icons/principles_icons/principle_tier_1.dds"

	visible = {
		has_dlc_feature = power_bloc_features
	}

	member_modifier = {
		country_spec_bloc_contagion_mult = -0.25
	}

	ai_weight = {
		value = 50
		if = {
			limit = { power_bloc ?= { any_power_bloc_member = { count >= 2 OR = { has_variable = spec_panic has_variable = spec_panic_cooldown } } } }
			add = 100
		}
		if = {
			limit = { has_variable = spec_panic }
			add = 50
		}
		if = {
			limit = { has_variable = spec_exch_stand var:spec_exch_stand >= 1 }
			add = 50
		}
		if = {
			limit = { has_variable = spec_exch_stand var:spec_exch_stand >= 2 }
			add = 50
		}
		if = {
			limit = { has_technology_researched = central_banking }
			add = 25
		}
		if = {
			limit = { has_strategy = ai_strategy_economic_imperialism }
			multiply = 1.5
		}
		if = {
			limit = { power_bloc ?= { num_power_bloc_members < 3 } }
			multiply = 0.5
		}
	}
}

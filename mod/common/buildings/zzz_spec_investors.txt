# investors favour building what is booming: each building's ai_value grows with its good's hype
INJECT:building_food_industry = {
	ai_value = {
		value = pmr_ai_value_food_industry
		if = {
			limit = {
				owner ?= { has_variable = spec_bhype_food_industry }
			}
			multiply = owner.var:spec_bhype_food_industry
		}
	}
}

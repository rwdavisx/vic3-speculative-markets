spec_sgui_rv_toggle = {
	scope = country
	is_shown = {
		NOT = { this = scope:target }
		OR = {
			spec_rv_chose = yes
			scope:target = { has_variable = spec_exch_stand }
		}
	}
	is_valid = { spec_can_rv_toggle = yes }
	effect = {
		if = {
			limit = { spec_rv_chose = yes spec_can_rv_end = yes }
			spec_rv_end = yes
		}
		else_if = {
			limit = { spec_rv_chose = no spec_can_rv_choose = yes }
			spec_rv_choose = yes
		}
	}
}

spec_sgui_rv_pick = {
	scope = country
	effect = {
		clear_variable_list = spec_rv_cands
		ordered_country = {
			limit = { is_country_alive = yes has_variable = spec_exch_stand NOT = { this = root } }
			order_by = spec_centre_order
			max = 999
			check_range_bounds = no
			save_scope_as = target
			root = {
				if = {
					limit = { spec_rv_chose = no spec_can_rv_choose = yes }
					add_to_variable_list = { name = spec_rv_cands target = scope:target }
				}
			}
		}
		ordered_country = {
			limit = { is_country_alive = yes has_variable = spec_exch_stand NOT = { this = root } }
			order_by = spec_centre_order
			max = 999
			check_range_bounds = no
			save_scope_as = target
			root = {
				if = {
					limit = { spec_rv_chose = no NOT = { spec_can_rv_choose = yes } }
					add_to_variable_list = { name = spec_rv_cands target = scope:target }
				}
			}
		}
	}
}

spec_lever_ui = {
	if = {
		limit = { has_variable = spec_restrain_$G$ }
		set_variable = { name = spec_ui_r_$G$ value = 2 }
	}
	else_if = {
		limit = {
			is_player = yes
			spec_lever_free = { KIND = restrain OTHER = encourage G = $G$ }
		}
		set_variable = { name = spec_ui_r_$G$ value = 1 }
	}
	else = {
		set_variable = { name = spec_ui_r_$G$ value = 0 }
	}
	if = {
		limit = { has_variable = spec_encourage_$G$ }
		set_variable = { name = spec_ui_e_$G$ value = 2 }
	}
	else_if = {
		limit = {
			is_player = yes
			spec_lever_free = { KIND = encourage OTHER = restrain G = $G$ }
		}
		set_variable = { name = spec_ui_e_$G$ value = 1 }
	}
	else = {
		set_variable = { name = spec_ui_e_$G$ value = 0 }
	}
	set_variable = { name = spec_ui_co_$G$ value = 0 }
	if = {
		limit = { var:spec_word_$G$ ?= 1 }
		set_variable = { name = spec_ui_co_$G$ value = -10 }
	}
	else_if = {
		limit = { var:spec_word_$G$ ?= 2 }
		set_variable = { name = spec_ui_co_$G$ value = -5 }
	}
	else_if = {
		limit = { var:spec_word_$G$ ?= 4 }
		set_variable = { name = spec_ui_co_$G$ value = 2 }
	}
	else_if = {
		limit = { var:spec_word_$G$ ?= 5 }
		set_variable = { name = spec_ui_co_$G$ value = 4 }
	}
	else_if = {
		limit = { var:spec_word_$G$ ?= 6 }
		set_variable = { name = spec_ui_co_$G$ value = 6 }
	}
	if = {
		limit = { has_variable = spec_restrain_$G$ }
		change_variable = { name = spec_ui_co_$G$ add = -10 }
	}
	else_if = {
		limit = { has_variable = spec_encourage_$G$ }
		change_variable = { name = spec_ui_co_$G$ add = 10 }
	}
}

spec_lever_flip = {
	if = {
		limit = { has_variable = spec_$KIND$_$G$ }
		remove_variable = spec_$KIND$_$G$
		remove_variable = spec_lever_pending_$G$
		remove_variable = spec_lever_grace_$G$
	}
	else = {
		set_variable = { name = spec_$KIND$_$G$ value = 1 }
		set_variable = spec_lever_pending_$G$
		set_variable = { name = spec_lever_grace_$G$ days = 7 }
	}
	# either way the good's levers wait (WAIT_MONTHS), after a few days to change our mind
	set_variable = { name = spec_lever_wait_$G$ value = 3 }
}

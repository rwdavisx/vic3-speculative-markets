# rival actions (actor scope, scope:target): their Exchange's size against ours, VS_LO to VS_HI. A larger rival
# costs more to act against or to shore up against; a larger Exchange's moves hit harder (spec_rv_heft, the inverse)
spec_rv_vs = {
	value = 1
	if = {
		limit = {
			has_variable = spec_centre_score
			scope:target ?= { has_variable = spec_centre_score }
		}
		value = scope:target.var:spec_centre_score
		divide = {
			value = var:spec_centre_score
			min = 1
		}
	}
	min = 0.75
	max = 1.33
}

spec_rv_heft = {
	value = 1
	divide = spec_rv_vs
}

# the yearly contest's stake, in the winner's scope against scope:spec_rv_l: an upset pays more, the expected less
spec_rv_stake = {
	value = 3
	if = {
		limit = { var:spec_exch_stand > scope:spec_rv_l.var:spec_exch_stand }
		divide = {
			value = var:spec_exch_stand
			subtract = scope:spec_rv_l.var:spec_exch_stand
			add = 1
		}
	}
	else_if = {
		limit = { var:spec_exch_stand < scope:spec_rv_l.var:spec_exch_stand }
		multiply = {
			value = scope:spec_rv_l.var:spec_exch_stand
			subtract = var:spec_exch_stand
			add = 1
		}
	}
}

# the contest's bar from scope:spec_rival_me's side against this: the score lead as a share of the higher score, then
# the pushes each way, from -1 (theirs) to 1 (ours)
spec_rv_rope = {
	value = spec_rv_me_eff
	subtract = spec_rv_it_eff
	divide = {
		value = spec_rv_it_eff
		min = spec_rv_me_eff
		min = 1
	}
	# blue chips (bluechips.PULL each as counted: in full up to CAP, half up to CAP2)
	if = {
		limit = { scope:spec_rival_me = { has_variable = spec_bc_n } }
		add = {
			value = 0
			add = { value = scope:spec_rival_me.var:spec_bc_n max = 5 }
			add = { value = scope:spec_rival_me.var:spec_bc_n subtract = 5 min = 0 max = 5 multiply = 0.5 }
			multiply = 0.03
		}
	}
	if = {
		limit = { has_variable = spec_bc_n }
		subtract = { value = spec_bc_counted multiply = 0.03 }
	}
	if = {
		limit = { scope:spec_rival_me = { is_target_in_variable_list = { name = spec_rv_push_1 target = prev } } }
		add = 0.1
	}
	if = {
		limit = { is_target_in_variable_list = { name = spec_rv_push_1 target = scope:spec_rival_me } }
		subtract = 0.1
	}
	if = {
		limit = { scope:spec_rival_me = { is_target_in_variable_list = { name = spec_rv_push_2 target = prev } } }
		add = 0.1
	}
	if = {
		limit = { is_target_in_variable_list = { name = spec_rv_push_2 target = scope:spec_rival_me } }
		subtract = 0.1
	}
	if = {
		limit = { scope:spec_rival_me = { is_target_in_variable_list = { name = spec_rv_push_3 target = prev } } }
		add = 0.1
	}
	if = {
		limit = { is_target_in_variable_list = { name = spec_rv_push_3 target = scope:spec_rival_me } }
		subtract = 0.1
	}
	if = {
		limit = { scope:spec_rival_me = { is_target_in_variable_list = { name = spec_rv_push_4 target = prev } } }
		add = 0.1
	}
	if = {
		limit = { is_target_in_variable_list = { name = spec_rv_push_4 target = scope:spec_rival_me } }
		subtract = 0.1
	}
	if = {
		limit = { scope:spec_rival_me = { is_target_in_variable_list = { name = spec_rv_push_5 target = prev } } }
		add = 0.1
	}
	if = {
		limit = { is_target_in_variable_list = { name = spec_rv_push_5 target = scope:spec_rival_me } }
		subtract = 0.1
	}
	min = -1
	max = 1
}

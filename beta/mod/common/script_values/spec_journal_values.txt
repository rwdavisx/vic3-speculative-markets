# the journal entry's stage: 6 Long Depression, 5 Panic, 4 Aftermath, 3 Mania, 2 Buoyant, 1 Steady, 0 Wary
spec_stage = {
	value = 1
	if = {
		limit = { has_variable = spec_ld var:spec_ld < 3 }
		value = 6
	}
	else_if = {
		limit = { has_variable = spec_panic }
		value = 5
	}
	else_if = {
		limit = { has_variable = spec_panic_cooldown }
		value = 4
	}
	else_if = {
		limit = { has_variable = spec_conf var:spec_conf >= 80 }
		value = 3
	}
	else_if = {
		limit = { has_variable = spec_conf var:spec_conf >= 60 }
		value = 2
	}
	else_if = {
		limit = { has_variable = spec_conf var:spec_conf < 40 }
		value = 0
	}
}
